/ the setup function runs once
void setup() {
// initialize digital pin 12 for the LED 1's output and pin 13 for the LED 2's output.
pinMode(12, OUTPUT);
pinMode(13, OUTPUT);
}

// the loop function runs over and over again forever
void loop() {
digitalWrite(12, HIGH); // turn on the LED 1
digitalWrite(13, LOW); // turn off the LED 2
delay(1000); // wait for a second
digitalWrite(12, LOW); // turn off the LED 1
digitalWrite(13, HIGH); // turn on the LED 2
delay(1000); // wait for a second
}
